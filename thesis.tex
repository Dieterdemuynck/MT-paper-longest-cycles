% Tube Graphs by Zamfirescu %%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\usetikzlibrary{calc}

\newcommand{\TubeGraph}[2]{%
  % #1 = p (cycle length)
  % #2 = l (edges in each connecting path)

  \def\p{#1}
  \def\l{#2}

  % Radii
  \def\R{2}
  \def\r{0.6}

  % Outer cycle vertices (a_i)
  \foreach \i in {1,...,\p} {
    \node[vertex]
      (A\i) at ({90 - 360/\p * (\i-1)}:\R) {};
  }

  % Inner cycle vertices (b_i)
  \foreach \i in {1,...,\p} {
    \node[vertex]
      (B\i) at ({90 - 360/\p * (\i-1)}:\r) {};
  }

  % Draw outer cycle
  \foreach \i [evaluate=\i as \next using int(\i+1)] in {1,...,\numexpr\p-1\relax} {
    \draw (A\i) -- (A\next);
  }

  % Draw inner cycle
  \foreach \i [evaluate=\i as \next using int(\i+1)] in {1,...,\numexpr\p-1\relax} {
    \draw (B\i) -- (B\next);
  }

  % Closing edges
  \ifodd\p
    \draw (A\p) -- (A1);
    \draw (B\p) -- (B1);
  \else
    \draw (A\p) -- (B1);
    \draw (B\p) -- (A1);
  \fi

  % Connecting paths H_i
  \foreach \i in {1,...,\p} {

    \def\prev{A\i}

    \foreach \j in {1,...,\numexpr\l-1\relax} {

      \pgfmathsetmacro{\t}{\j/\l}

      \node[vertex]
        (H\i-\j)
        at ($(A\i)!\t!(B\i)$) {};

      \draw (\prev) -- (H\i-\j);

      \xdef\prev{H\i-\j}
    }

    \draw (\prev) -- (B\i);
  }
}

% Extended Tube Graphs
\newcommand{\ExtendedTubeGraph}[3]{%
  % #1 = p (cycle length)
  % #2 = l (edges in each connecting path)
  % #3 = d (edges in each cycle edge)

  \pgfmathtruncatemacro{\P}{#1}
  \pgfmathtruncatemacro{\L}{#2}
  \pgfmathtruncatemacro{\D}{#3}

  % Radii
  \def\R{2}
  \def\r{1}

  % Outer cycle vertices
  \foreach \i in {1,...,\P}{
    \node[vertex]
      (A\i)
      at ({90-360/\P*(\i-1)}:\R) {};
  }

  % Inner cycle vertices
  \foreach \i in {1,...,\P}{
    \node[vertex]
      (B\i)
      at ({90-360/\P*(\i-1)}:\r) {};
  }

  % Outer cycle
  \foreach \i in {1,...,\P}{
    \pgfmathtruncatemacro{\next}{mod(\i,\P)+1}

    \def\prev{A\i}

    \foreach \j in {1,...,\numexpr\D-1\relax}{
      \pgfmathsetmacro{\t}{\j/\D}

      \node[vertex]
        (DA\i-\j)
        at ($(A\i)!\t!(A\next)$) {};

      \draw (\prev) -- (DA\i-\j);

      \xdef\prev{DA\i-\j}
    }

    \ifodd\P
      \draw (\prev) -- (A\next);
    \else
      \ifnum\i=\P
        \draw (\prev) -- (B1);
      \else
        \draw (\prev) -- (A\next);
      \fi
    \fi
  }

  % Inner cycle
  \foreach \i in {1,...,\P}{
    \pgfmathtruncatemacro{\next}{mod(\i,\P)+1}

    \def\prev{B\i}

    \foreach \j in {1,...,\numexpr\D-1\relax}{
      \pgfmathsetmacro{\t}{\j/\D}

      \node[vertex]
        (DB\i-\j)
        at ($(B\i)!\t!(B\next)$) {};

      \draw (\prev) -- (DB\i-\j);

      \xdef\prev{DB\i-\j}
    }

    \ifodd\P
      \draw (\prev) -- (B\next);
    \else
      \ifnum\i=\P
        \draw (\prev) -- (A1);
      \else
        \draw (\prev) -- (B\next);
      \fi
    \fi
  }

  % Connecting paths
  \foreach \i in {1,...,\P}{
    \def\prev{A\i}

    \foreach \j in {1,...,\numexpr\L-1\relax}{
      \pgfmathsetmacro{\t}{\j/\L}

      \node[vertex]
        (H\i-\j)
        at ($(A\i)!\t!(B\i)$) {};

      \draw (\prev) -- (H\i-\j);

      \xdef\prev{H\i-\j}
    }

    \draw (\prev) -- (B\i);
  }
}
