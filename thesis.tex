\usetikzlibrary{calc}

\newcommand{\TubeGraph}[2]{%
  % #1 = p (cycle length)
  % #2 = l (edges in each connecting path)

  \def\p{#1}
  \def\l{#2}

  % Radii
  \def\R{2}
  \def\r{0.6}

  % Outer cycle vertices (a_i)
  \foreach \i in {1,...,\p} {
    \node[vertex]
      (A\i) at ({90 - 360/\p * (\i-1)}:\R) {};
  }

  % Inner cycle vertices (b_i)
  \foreach \i in {1,...,\p} {
    \node[vertex]
      (B\i) at ({90 - 360/\p * (\i-1)}:\r) {};
  }

  % Draw outer cycle
  \foreach \i [evaluate=\i as \next using int(\i+1)] in {1,...,\numexpr\p-1\relax} {
    \draw (A\i) -- (A\next);
  }

  % Draw inner cycle
  \foreach \i [evaluate=\i as \next using int(\i+1)] in {1,...,\numexpr\p-1\relax} {
    \draw (B\i) -- (B\next);
  }

  % Closing edges
  \ifodd\p
    \draw (A\p) -- (A1);
    \draw (B\p) -- (B1);
  \else
    \draw (A\p) -- (B1);
    \draw (B\p) -- (A1);
  \fi

  % Connecting paths H_i
  \foreach \i in {1,...,\p} {

    \def\prev{A\i}

    \foreach \j in {1,...,\numexpr\l-1\relax} {

      \pgfmathsetmacro{\t}{\j/\l}

      \node[vertex]
        (H\i-\j)
        at ($(A\i)!\t!(B\i)$) {};

      \draw (\prev) -- (H\i-\j);

      \xdef\prev{H\i-\j}
    }

    \draw (\prev) -- (B\i);
  }
}

% Extended Tube Graphs
\newcommand{\ExtendedTubeGraph}[3]{%
  % #1 = p (cycle length)
  % #2 = l (edges in each connecting path)
  % #3 = d (edges in each cycle edge)

  \pgfmathtruncatemacro{\P}{#1}
  \pgfmathtruncatemacro{\L}{#2}
  \pgfmathtruncatemacro{\D}{#3}

  % Radii
  \def\R{2}
  \def\r{1}

  % Outer cycle vertices
  \foreach \i in {1,...,\P}{
    \node[vertex]
      (A\i)
      at ({90-360/\P*(\i-1)}:\R) {};
  }

  % Inner cycle vertices
  \foreach \i in {1,...,\P}{
    \node[vertex]
      (B\i)
      at ({90-360/\P*(\i-1)}:\r) {};
  }

  % Outer cycle
  \foreach \i in {1,...,\P}{
    \pgfmathtruncatemacro{\next}{mod(\i,\P)+1}

    \def\prev{A\i}

    \foreach \j in {1,...,\numexpr\D-1\relax}{
      \pgfmathsetmacro{\t}{\j/\D}

      \node[vertex]
        (DA\i-\j)
        at ($(A\i)!\t!(A\next)$) {};

      \draw (\prev) -- (DA\i-\j);

      \xdef\prev{DA\i-\j}
    }

    \ifodd\P
      \draw (\prev) -- (A\next);
    \else
      \ifnum\i=\P
        \draw (\prev) -- (B1);
      \else
        \draw (\prev) -- (A\next);
      \fi
    \fi
  }

  % Inner cycle
  \foreach \i in {1,...,\P}{
    \pgfmathtruncatemacro{\next}{mod(\i,\P)+1}

    \def\prev{B\i}

    \foreach \j in {1,...,\numexpr\D-1\relax}{
      \pgfmathsetmacro{\t}{\j/\D}

      \node[vertex]
        (DB\i-\j)
        at ($(B\i)!\t!(B\next)$) {};

      \draw (\prev) -- (DB\i-\j);

      \xdef\prev{DB\i-\j}
    }

    \ifodd\P
      \draw (\prev) -- (B\next);
    \else
      \ifnum\i=\P
        \draw (\prev) -- (A1);
      \else
        \draw (\prev) -- (B\next);
      \fi
    \fi
  }

  % Connecting paths
  \foreach \i in {1,...,\P}{
    \def\prev{A\i}

    \foreach \j in {1,...,\numexpr\L-1\relax}{
      \pgfmathsetmacro{\t}{\j/\L}

      \node[vertex]
        (H\i-\j)
        at ($(A\i)!\t!(B\i)$) {};

      \draw (\prev) -- (H\i-\j);

      \xdef\prev{H\i-\j}
    }

    \draw (\prev) -- (B\i);
  }
}

\begin{abstract}
This thesis studies the intersection of longest cycles in graphs, continuing a line of research initiated by van Aardt et al.\ and extended by Carol T.\ Zamfirescu. Van Aardt et al.\ introduced the parameter $\alpha(k)$, the smallest fraction of vertices whose removal destroys all cycles of length $k$ in every graph of circumference $k$, and showed if for a value $k$ there exists a 2-connected graph of circumference $k$ in which no single vertex meets every longest cycle, then this value could be directly determined as $\alpha(k) = \frac{1}{k}$. This question was resolved for all $k$ except $k \in \{10, 11, 13, 16\}$. Zamfirescu subsequently settled the cases $k = 10, 13, 16$, leaving $k = 11$ as the only open case. This thesis investigates this remaining case (RQ\ref{rq1}). Independently, Zamfirescu asked a related question (RQ\ref{rq2}): for which values of $k$ does the existence of a $k$-cycle in every vertex-deleted subgraph of a planar graph of minimum degree at least 4 require the graph to contain a longer cycle?

We approach both questions using computer-assisted graph theory. We design and implement a bitset-optimized depth-first cycle-enumeration algorithm that computes a graph's circumference together with the intersection of its longest (or fixed-length) cycles. These algorithms are combined with the generators \texttt{geng}, \texttt{GenerateUHG}, and \texttt{plantri}, and a list of pre-generated graphs, to exhaustively search restricted classes of 2-connected graphs.

For RQ\ref{rq1}, exhaustive search finds no counterexample among 2-connected graphs of order up to 22 with maximum degree at most 3, nor for the further (order, upper bound for maximum degree) combinations of $(13,6)$, $(14,5)$, and $(15,4)$, nor among planar 2-connected graphs of order 13, the last of which have at least four vertices meeting every 11-cycle. RQ\ref{rq1} therefore remains open. Based on the results however, we generalize this into an infinite family (Theorem~\ref{theorem:single-vertex-intersection}). for every order $n \ge 13$, there exists a 2-connected graph of circumference 11 in which exactly one vertex meets every (longest) 11-cycle.

For RQ\ref{rq2}, exhaustive search over 2-connected planar graphs of minimum degree at least 4, extended with a supplied graph list to order 20, finds no negative example: the statement of RQ2 holds for every graph checked. In parallel, we generalize the gadget underlying Zamfirescu's construction $G_{p,\ell}$ into a new family $G_{p,\ell,D}$ which finds new values $k$ for specific orders $n$ which are negative examples to the question RQ\ref{rq2}~(Theorem~\ref{theorem:new-vals-rq2}).
\end{abstract}
